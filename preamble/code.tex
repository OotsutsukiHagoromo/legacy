% ============================ minted ===============================

\usepackage{minted}

\setminted{
  style=friendly,
  linenos,
  numbersep=8pt,
  breaklines,
  autogobble,
  tabsize=4%,
}

% Новые окружения
% \begin{...} ... \end{...}
\newminted{html}{}  % htmlcode
\newminted{latex}{} % latexcode

% Новые команды для внутристрочных вставок
\newmintinline[htmlinline]{html}{}   % \htmlinline{...}
\newmintinline[latexinline]{latex}{} % \latexinline{...}
